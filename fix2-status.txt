    13	     * Rank eligible nodes deterministically. The provider is still queried immediately

    68	    private function hasCapacity(mixed $capacity, mixed $allocated, mixed $reserved, int $required): bool
    69	    {
    70	        // Placement is fail-closed. A missing capacity must never turn an unknown node
    71	        // into an unlimited one; the snapshot refresher fills CPU from Wings when an
    72	        // administrator has not supplied an explicit value.
    73	        if ($capacity === null || (int) $capacity <= 0) {
    74	            return false;
    75	        }
    76	
    77	        // CtrlPanel/Pterodactyl use zero as "unlimited", not "no resource".
    78	        // Reserve the complete node capacity so an unlimited server can only be
    79	        // provisioned on an otherwise empty node.
    80	        if ($required === 0) {
    81	            $required = (int) $capacity;
    82	        }
    83	
    84	        return (int) $capacity - max(0, (int) $allocated) - max(0, (int) $reserved) >= $required;
    85	    }
