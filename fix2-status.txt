async def open_backend(
    info: dict,
    proxy_line: bytes = b""
):
    host = info["backend"]["host"]
    port = int(info["backend"]["port"])

    context = JAVA_TRANSPARENT_CONTEXT.get()

    use_transparent = False
    source_ip = ""
    source_port = 0
    fqdn = ""

    if context:
        try:
            fqdn = str(context[0])
            source_ip = str(context[1])
            source_port = int(context[2])

            use_transparent = (
                fqdn == JAVA_TRANSPARENT_DOMAIN
                and bool(JAVA_TRANSPARENT_BACKEND_HOST)
                and source_ip
                and source_ip != "0.0.0.0"
                and source_ip != "::"
                and source_ip != "::1"
                and not source_ip.startswith("127.")
                and ":" not in source_ip
                and source_port > 0
            )

        except Exception:
            use_transparent = False

    if use_transparent:

        target_host = JAVA_TRANSPARENT_BACKEND_HOST

        sock = socket.socket(
            socket.AF_INET,
            socket.SOCK_STREAM
        )

        try:

            sock.setsockopt(
                socket.SOL_IP,
                19,
                1
            )

            sock.setblocking(False)

            sock.bind(
                (
                    source_ip,
                    source_port
                )
            )

            loop = asyncio.get_running_loop()

            await asyncio.wait_for(
                loop.sock_connect(
                    sock,
                    (
                        target_host,
                        port
                    )
                ),
                timeout=BACKEND_CONNECT_TIMEOUT
            )

            reader, writer = await asyncio.open_connection(
                sock=sock
            )

            LOG.info(
                "JAVA_TRANSPARENT_CONNECT "
                "fqdn=%s source=%s:%s backend=%s:%s",
                fqdn,
                source_ip,
                source_port,
                target_host,
                port,
            )

            return reader, writer

        except Exception:

            sock.close()

            LOG.exception(
                "JAVA_TRANSPARENT_CONNECT_FAILED "
                "fqdn=%s source=%s:%s backend=%s:%s",
                fqdn,
                source_ip,
                source_port,
                target_host,
                port,
            )

            raise

    reader, writer = await asyncio.wait_for(
        asyncio.open_connection(
            host,
            port
        ),
        timeout=BACKEND_CONNECT_TIMEOUT
    )

    if proxy_line:
        writer.write(proxy_line)
        await writer.drain()

    return reader, writer

async def handle_java(
    reader: asyncio.StreamReader,
    writer: asyncio.StreamWriter,
    api: WakeApi,
    tracker: ActivityTracker
):
    # FIX2_V41_REAL_TCP_ACCEPT
    _accept_peer = writer.get_extra_info("peername") or ("unknown", 0)
    _accept_local = writer.get_extra_info("sockname") or ("unknown", 0)
    LOG.info(
        "JAVA_TCP_ACCEPT source=%s:%s local=%s:%s",
        _accept_peer[0],
        _accept_peer[1] if len(_accept_peer) > 1 else 0,
        _accept_local[0],
        _accept_local[1] if len(_accept_local) > 1 else 0,
    )
    peer = (
        writer.get_extra_info("peername")
        or ("0.0.0.0", 0)
    )

    client_ip = str(peer[0])
    client_port = int(peer[1]) if len(peer) > 1 else 0

    local = (
        writer.get_extra_info("sockname")
        or ("0.0.0.0", 0)
    )

    local_ip = str(local[0])
    local_port = int(local[1]) if len(local) > 1 else 0

    track_key = f"java:{id(writer)}"

    fqdn = None
    correlation = None
    backend_writer = None

    try:
        handshake_raw, payload = await asyncio.wait_for(
            read_packet(
                reader,
                4096
            ),
            timeout=JAVA_INITIAL_TIMEOUT
        )

        (
            protocol,
            fqdn,
            requested_port,
            next_state
        ) = parse_handshake(payload)

        info = await api.resolve_java(
            fqdn
        )

        if not info:
            LOG.info(
                "unknown Java host %s from %s",
                fqdn,
                client_ip
            )
            return

        proxy_line = build_java_proxy_line(
            fqdn,
            client_ip,
            client_port,
            local_ip,
            local_port,
        )

        # -------------------------------------------------
        # STATUS / SERVER LIST
        # Darf niemals einen Wake auslösen.
        # -------------------------------------------------
        if next_state == 1:

            try:
                br, bw = await open_backend(info, proxy_line)

            except Exception:

                await local_status(
                    reader,
                    writer,
                    info,
                    protocol
                )

                return

            try:
                await proxy_java(
                    reader,
                    writer,
                    br,
                    bw,
                    handshake_raw,
                    tracker,
                    track_key
                )

            finally:
                bw.close()

                try:
                    await bw.wait_closed()
                except Exception:
                    pass

            return

        # -------------------------------------------------
        # LOGIN
        #
        # Zuerst Backend direkt testen.
        # Wenn es läuft, einfach normal durchreichen.
        # -------------------------------------------------
        try:
            backend_reader, backend_writer = (
                await open_backend(info, proxy_line)
            )

            initial = handshake_raw

        except Exception:

            # Backend offline:
            # Erst jetzt darf ein möglicher Wake geprüft werden.
            if not info.get("wake_allowed"):
                return

            # Ein bloßer Handshake reicht NICHT zum Aufwecken.
            # Wir verlangen ein echtes Login-Start-Paket.
            login_raw, login_payload = (
                await asyncio.wait_for(
                    read_packet(
                        reader,
                        4096
                    ),
                    timeout=JAVA_INITIAL_TIMEOUT
                )
            )

            username = parse_login_start(
                login_payload
            )

            initial = (
                handshake_raw
                + login_raw
            )

            LOG.info(
                "valid Java login attempt for %s from %s user=%s",
                fqdn,
                client_ip,
                username
            )

            endpoint = (
                f"{fqdn}:{requested_port}"
            )

            wake = await api.wake(
                info,
                endpoint,
                client_ip
            )

            correlation = wake.get(
                "correlation_id"
            )

            wait_seconds = max(
                5,
                min(
                    MAX_WAKE_WAIT,
                    int(
                        wake.get(
                            "wait_seconds"
                        )
                        or MAX_WAKE_WAIT
                    )
                )
            )

            LOG.info(
                "wake requested for %s; "
                "waiting up to %ss",
                fqdn,
                wait_seconds
            )

            deadline = (
                time.monotonic()
                + wait_seconds
            )

            last_error = None

            while time.monotonic() < deadline:

                try:
                    info = (
                        await api.resolve_java(
                            fqdn,
                            cached=False
                        )
                        or info
                    )

                    (
                        backend_reader,
                        backend_writer
                    ) = await open_backend(info, proxy_line)

                    LOG.info(
                        "backend ready for %s",
                        fqdn
                    )

                    break

                except Exception as exc:
                    last_error = exc

                    await asyncio.sleep(
                        WAKE_POLL_INTERVAL
                    )

            else:

                if correlation:
                    await api.complete(
                        correlation,
                        "TIMEOUT",
                        (
                            "Backend did not become "
                            f"reachable: {last_error}"
                        )
                    )

                LOG.warning(
                    "wake timeout for %s",
                    fqdn
                )

                return

        await tracker.add(
            track_key,
            fqdn
        )

        if correlation:

            await api.complete(
                correlation,
                "COMPLETED"
            )

        await proxy_java(
            reader,
            writer,
            backend_reader,
            backend_writer,
            initial,
            tracker,
            track_key
        )

    except (
        asyncio.IncompleteReadError,
        ConnectionError,
        ProtocolError,
        asyncio.TimeoutError
    ):
        pass

    except Exception:

        LOG.exception(
            "Java gateway error for %s",
            fqdn or client_ip
        )

        if correlation:

            await api.complete(
                correlation,
                "FAILED",
                "Gateway proxy error"
            )

    finally:

        await tracker.remove(
            track_key
        )

        if backend_writer:

            backend_writer.close()

            try:
                await backend_writer.wait_closed()
            except Exception:
                pass

        writer.close()

        try:
            await writer.wait_closed()
        except Exception:
            pass

@dataclass
class BedrockSession:
    addr: tuple
    fqdn: str
    info: dict
    listener: "BedrockListener"
    api: WakeApi
    tracker: ActivityTracker
    transport: Optional[asyncio.DatagramTransport] = None
    protocol: Optional["BedrockBackend"] = None
    last_seen: float = field(default_factory=time.monotonic)
    backend_seen: bool = False
    waking: bool = False
    correlation: Optional[str] = None
    first_payload: Optional[bytes] = None
    track_key: str = ""

    async def ensure_backend(self):
        if self.transport:
            return
        loop = asyncio.get_running_loop()
        host = self.info["backend"]["host"]
        port = int(self.info["backend"]["port"])
        protocol = BedrockBackend(self)
        transport, _ = await loop.create_datagram_endpoint(lambda: protocol, remote_addr=(host, port))
        self.transport = transport
        self.protocol = protocol

    async def send(self, data: bytes):
        self.last_seen = time.monotonic()
        await self.ensure_backend()
        self.transport.sendto(data)

    async def wake_and_retry(self):
        if self.waking:
            return
        self.waking = True
        try:
            wake = await self.api.wake(self.info, f"{self.info['fqdn']}:{self.listener.port}", self.addr[0])
            self.correlation = wake.get("correlation_id")
            deadline = time.monotonic() + max(5, min(MAX_WAKE_WAIT, int(wake.get("wait_seconds") or MAX_WAKE_WAIT)))
            while time.monotonic() < deadline and not self.backend_seen:
                if self.first_payload and self.transport:
                    self.transport.sendto(self.first_payload)
                await asyncio.sleep(0.8)
            if self.backend_seen:
                if self.correlation:
                    await self.api.complete(self.correlation, "COMPLETED")
            elif self.correlation:
                await self.api.complete(self.correlation, "TIMEOUT", "Bedrock backend did not answer")
        except Exception:
            LOG.exception("Bedrock wake failed for %s", self.info.get("fqdn"))
            if self.correlation:
                await self.api.complete(self.correlation, "FAILED", "Bedrock gateway wake error")
        finally:
            self.waking = False

    async def close(self):
        await self.tracker.remove(self.track_key)
        if self.transport:
            self.transport.close()
            self.transport = None


class BedrockBackend(asyncio.DatagramProtocol):
    def __init__(self, session: BedrockSession):
        self.session = session

    def datagram_received(self, data, addr):
        self.session.backend_seen = True
        self.session.last_seen = time.monotonic()
        if self.session.listener.transport:
            self.session.listener.transport.sendto(data, self.session.addr)
        asyncio.create_task(self.session.tracker.mark_active(self.session.track_key))


class BedrockListener(asyncio.DatagramProtocol):
    def __init__(self, port: int, api: WakeApi, tracker: ActivityTracker):
        self.port = port
        self.api = api
        self.tracker = tracker
        self.transport = None
        self.sessions = {}
        self.info = None
        self.last_resolve = 0.0

    def connection_made(self, transport):
        self.transport = transport

    async def resolve(self):
        now = time.monotonic()
        if self.info and now - self.last_resolve < CACHE_TTL:
            return self.info
        self.info = await self.api.resolve_bedrock_port(self.port, cached=False)
        self.last_resolve = now
        return self.info

    def datagram_received(self, data: bytes, addr):
        asyncio.create_task(self.handle(data, addr))

    async def handle(self, data: bytes, addr):
        info = await self.resolve()
        if not info:
            return
        packet_id = data[0] if data else -1
        if packet_id in (0x01, 0x02) and len(data) >= 9:
            # Ping never wakes a sleeping server. If a live backend session exists,
            # it will be used by normal traffic; status can safely be answered here.
            self.send_pong(data, addr, info)
            return

        key = addr
        session = self.sessions.get(key)
        if not session:
            track_key = f"bedrock:{self.port}:{addr[0]}:{addr[1]}"
            session = BedrockSession(addr, info["fqdn"], info, self, self.api, self.tracker, track_key=track_key)
            self.sessions[key] = session
            await self.tracker.add(track_key, info["fqdn"])
        session.first_payload = session.first_payload or data
        await session.send(data)

        # A RakNet connection request is a real join attempt. Give the running
        # backend a short chance to answer, then wake if it stays silent.
        if packet_id in (0x05, 0x07) and not session.backend_seen and info.get("wake_allowed"):
            await asyncio.sleep(BACKEND_CONNECT_TIMEOUT)
            if not session.backend_seen:
                asyncio.create_task(session.wake_and_retry())

    def send_pong(self, ping: bytes, addr, info: dict):
        profile = info.get("status_profile") or {}

        ping_time = (
            ping[1:9]
            if len(ping) >= 9
            else b"\x00" * 8
        )

        protocol = int(
            profile.get("protocol") or 0
        )

        version = str(
            profile.get("version") or "Minecraft"
        )

        max_players = int(
            profile.get("max_players") or 20
        )

        online_players = max(
            0,
            int(profile.get("online_players") or 0)
        )

        motd = str(
            profile.get("motd")
            or info.get("fqdn")
            or "Minecraft Server"
        ).replace(";", ",")

        runtime = str(
            info.get("runtime_state") or "unknown"
        ).lower()

        if runtime in ("sleeping", "offline"):

            online_players = 0

            sub_motd = (
                "Server schläft - "
                "Beitreten startet ihn"
            )

        elif runtime == "starting":

            sub_motd = (
                "Server startet gerade - "
                "Bitte kurz warten"
            )

        elif runtime == "stopping":

            sub_motd = (
                "Server wird gestoppt"
            )

        elif runtime == "running":

            sub_motd = "Server online"

        else:

            sub_motd = (
                "Server wird vorbereitet"
            )

        status = (
            f"MCPE;"
            f"{motd};"
            f"{protocol};"
            f"{version};"
            f"{online_players};"
            f"{max_players};"
            f"{SERVER_GUID};"
            f"{sub_motd};"
            f"Survival;"
            f"1;"
            f"{self.port};"
            f"{self.port}"
        )

        raw = status.encode("utf-8")[:1600]

        packet = (
            b"\x1c"
            + ping_time
            + SERVER_GUID.to_bytes(8, "big")
            + MAGIC
            + struct.pack(">H", len(raw))
            + raw
        )

        self.transport.sendto(
            packet,
            addr
        )

    async def cleanup(self):
        while True:
            await asyncio.sleep(15)
            cutoff = time.monotonic() - BEDROCK_SESSION_TTL
            stale = [key for key, sess in self.sessions.items() if sess.last_seen < cutoff]
            for key in stale:
                sess = self.sessions.pop(key)
                await sess.close()

async def request(self, endpoint: str, payload: dict, timeout: float = 5.0):
        return await asyncio.to_thread(self._request_sync, endpoint, payload, timeout)

    async def resolve_java(self, fqdn: str, cached=True):
        key = ("java", fqdn)
        now = time.monotonic()
        if cached and key in self.cache and self.cache[key][0] > now:
            return self.cache[key][1]
        status, data = await self.request("resolve", {"fqdn": fqdn, "edition": "java"})
        if status == 200 and data.get("known"):
            self.cache[key] = (now + CACHE_TTL, data)
            return data
        return None

    async def resolve_bedrock_port(self, port: int, cached=True):
        key = ("bedrock", port)
        now = time.monotonic()
        if cached and key in self.cache and self.cache[key][0] > now:
            return self.cache[key][1]
        status, data = await self.request("resolve-port", {"port": port, "edition": "bedrock"})
        if status == 200 and data.get("known"):
            self.cache[key] = (now + CACHE_TTL, data)
            return data
        return None

    async def wake(self, info: dict, endpoint: str, client_ip: str):
        status, data = await self.request("wake", {
            "fqdn": info["fqdn"],
            "edition": info["edition"],
            "endpoint": endpoint,
            "client_ip": client_ip,
        }, timeout=8)
        if status == 202 and data.get("accepted"):
            return data
        raise RuntimeError(data.get("message") or f"wake API returned HTTP {status}")

    async def complete(self, correlation_id: str, state: str, reason: Optional[str] = None):
        payload = {"correlation_id": correlation_id, "state": state}
        if reason:
            payload["reason"] = reason[:500]
        try:
            await self.request("complete", payload, timeout=5)
        except Exception:
            LOG.exception("unable to complete wake session %s", correlation_id)

    async def activity(self, fqdn: str, connections: int, active: int, window: int):
        try:
            status, _ = await self.request("activity", {
                "fqdn": fqdn,
                "observed_at": int(time.time()),
                "connections": connections,
                "active_connections": min(active, connections),
                "window_seconds": max(15, min(300, window)),
            }, timeout=5)
            if status not in (200, 409):
                LOG.warning("activity for %s returned HTTP %s", fqdn, status)
        except Exception:
            LOG.exception("activity report failed for %s", fqdn)

    async def health(self):
        status, data = await self.request("health", {}, timeout=5)
        if status != 200 or not data.get("ok"):
            raise RuntimeError(f"wake API health failed: HTTP {status} {data}")
        return data


class ActivityTracker:
    def __init__(self, api: WakeApi):
        self.api = api
        self.sessions = {}
        self.last_reported = set()
        self.lock = asyncio.Lock()

    async def add(self, key: str, fqdn: str):
        async with self.lock:
            self.sessions[key] = {"fqdn": fqdn, "active": False}

    async def mark_active(self, key: str):
        async with self.lock:
            if key in self.sessions:
                self.sessions[key]["active"] = True

    async def remove(self, key: str):
        async with self.lock:
            self.sessions.pop(key, None)

    async def snapshot(self):
        async with self.lock:
            out = {}
            for item in self.sessions.values():
                row = out.setdefault(item["fqdn"], [0, 0])
                row[0] += 1
                row[1] += int(item["active"])
            return out

    async def reporter(self):
        while True:
            await asyncio.sleep(ACTIVITY_INTERVAL)
            current = await self.snapshot()
            names = set(current) | self.last_reported
            for fqdn in names:
                conn, active = current.get(fqdn, (0, 0))
                await self.api.activity(fqdn, conn, active, ACTIVITY_INTERVAL)
            self.last_reported = set(current)


class ProtocolError(Exception):
    pass


def decode_varint_from_bytes(buf: bytes, offset=0):
    value = 0
    position = 0
    i = offset
    while i < len(buf):
        b = buf[i]
        value |= (b & 0x7F) << position
        i += 1
        if not (b & 0x80):
            return value, i
        position += 7
        if position >= 35:
            raise ProtocolError("VarInt too large")
    raise ProtocolError("Incomplete VarInt")


def encode_varint(value: int):
    out = bytearray()
    value &= 0xFFFFFFFF
    while True:
        temp = value & 0x7F
        value >>= 7
        if value:
            temp |= 0x80
        out.append(temp)
        if not value:
            return bytes(out)


def encode_mc_string(value: str):
    raw = value.encode("utf-8")
    return encode_varint(len(raw)) + raw


async def read_varint(reader: asyncio.StreamReader):
    value = 0
    position = 0
    raw = bytearray()
    while True:
        b = (await reader.readexactly(1))[0]
        raw.append(b)
        value |= (b & 0x7F) << position
        if not (b & 0x80):
            return value, bytes(raw)
        position += 7
        if position >= 35:
            raise ProtocolError("VarInt too large")


async def read_packet(reader: asyncio.StreamReader, max_size=2_000_000):
    length, prefix = await read_varint(reader)
    if length < 0 or length > max_size:
        raise ProtocolError("Invalid packet length")
    payload = await reader.readexactly(length)
    return prefix + payload, payload


def parse_handshake(payload: bytes):
    packet_id, pos = decode_varint_from_bytes(payload, 0)
    if packet_id != 0:
        raise ProtocolError("First packet is not handshake")
    protocol, pos = decode_varint_from_bytes(payload, pos)
    slen, pos = decode_varint_from_bytes(payload, pos)
    if slen < 1 or slen > 255 or pos + slen + 2 > len(payload):
        raise ProtocolError("Invalid handshake hostname")
    host_raw = payload[pos:pos+slen].decode("utf-8", errors="replace")
    host = host_raw.split("\x00", 1)[0].lower().rstrip(".")
    pos += slen
    server_port = struct.unpack(">H", payload[pos:pos+2])[0]
    pos += 2
    next_state, pos = decode_varint_from_bytes(payload, pos)
    if next_state not in (1, 2):
        raise ProtocolError("Unsupported next state")
    return protocol, host, server_port, next_state


def parse_login_start(payload: bytes):
    packet_id, pos = decode_varint_from_bytes(
        payload,
        0
    )

    if packet_id != 0:
        raise ProtocolError(
            "First login packet is not Login Start"
        )

    name_len, pos = decode_varint_from_bytes(
        payload,
        pos
    )

    if (
        name_len < 1
        or name_len > 16
        or pos + name_len > len(payload)
    ):
        raise ProtocolError(
            "Invalid Login Start username"
        )

    try:
        username = payload[
            pos:pos + name_len
        ].decode("ascii")
    except UnicodeDecodeError:
        raise ProtocolError(
            "Invalid username encoding"
        )

    allowed = (
        "abcdefghijklmnopqrstuvwxyz"
        "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
        "0123456789_"
    )

    if not all(c in allowed for c in username):
        raise ProtocolError(
            "Invalid Minecraft username"
        )

    return username


async def open_backend(
    info: dict,
    proxy_line: bytes = b""
):
    host = info["backend"]["host"]
    port = int(info["backend"]["port"])

    context = JAVA_TRANSPARENT_CONTEXT.get()

    use_transparent = False
    source_ip = ""
    source_port = 0
    fqdn = ""

    if context:
        try:
            fqdn = str(context[0])
            source_ip = str(context[1])
            source_port = int(context[2])

            use_transparent = (
                fqdn == JAVA_TRANSPARENT_DOMAIN
                and bool(JAVA_TRANSPARENT_BACKEND_HOST)
                and source_ip
                and source_ip != "0.0.0.0"
                and source_ip != "::"
                and source_ip != "::1"
                and not source_ip.startswith("127.")
                and ":" not in source_ip
                and source_port > 0
            )

        except Exception:
            use_transparent = False

    if use_transparent:

        target_host = JAVA_TRANSPARENT_BACKEND_HOST

        sock = socket.socket(
            socket.AF_INET,
            socket.SOCK_STREAM
        )

        try:

            sock.setsockopt(
                socket.SOL_IP,
                19,
                1
            )

            sock.setblocking(False)

            sock.bind(
                (
                    source_ip,
                    source_port
                )
            )

            loop = asyncio.get_running_loop()

            await asyncio.wait_for(
                loop.sock_connect(
                    sock,
                    (
                        target_host,
                        port
                    )
                ),
                timeout=BACKEND_CONNECT_TIMEOUT
            )

            reader, writer = await asyncio.open_connection(
                sock=sock
            )

            LOG.info(
                "JAVA_TRANSPARENT_CONNECT "
                "fqdn=%s source=%s:%s backend=%s:%s",
                fqdn,
                source_ip,
                source_port,
                target_host,
                port,
            )

            return reader, writer

        except Exception:

            sock.close()

            LOG.exception(
                "JAVA_TRANSPARENT_CONNECT_FAILED "
                "fqdn=%s source=%s:%s backend=%s:%s",
                fqdn,
                source_ip,
                source_port,
                target_host,
                port,
            )

            raise

    reader, writer = await asyncio.wait_for(
        asyncio.open_connection(
            host,
            port
        ),
        timeout=BACKEND_CONNECT_TIMEOUT
    )

    if proxy_line:
        writer.write(proxy_line)
        await writer.drain()

    return reader, writer

async def relay(src: asyncio.StreamReader, dst: asyncio.StreamWriter, active_cb=None):
    total = 0
    try:
        while True:
            data = await src.read(65536)
            if not data:
                break
            dst.write(data)
            await dst.drain()
            total += len(data)
            if active_cb and total >= 96:
                await active_cb()
                active_cb = None
    except (ConnectionError, asyncio.IncompleteReadError, BrokenPipeError):
        pass
    finally:
        try:
            dst.write_eof()
        except Exception:
            pass


async def proxy_java(reader, writer, backend_reader, backend_writer, initial, tracker, track_key):
    backend_writer.write(initial)
    await backend_writer.drain()
    async def active():
        await tracker.mark_active(track_key)
    tasks = [
        asyncio.create_task(relay(reader, backend_writer, active)),
        asyncio.create_task(relay(backend_reader, writer, active)),
    ]
    done, pending = await asyncio.wait(tasks, return_when=asyncio.FIRST_COMPLETED)
    for task in pending:
        task.cancel()
    await asyncio.gather(*pending, return_exceptions=True)
